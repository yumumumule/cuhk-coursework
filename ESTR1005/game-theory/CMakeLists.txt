cmake_minimum_required(VERSION 3.16)
project(estr1005_nash VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(nash_core OBJECT src/nash.c)
target_include_directories(nash_core PUBLIC include)
if(MSVC)
    target_compile_options(nash_core PRIVATE /W4)
else()
    target_compile_options(nash_core PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(nash_core PUBLIC m)
endif()
add_executable(nash src/main.c)
target_link_libraries(nash PRIVATE nash_core)
if(MSVC)
    target_compile_options(nash PRIVATE /W4)
else()
    target_compile_options(nash PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME solver_regression
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_solver.py $<TARGET_FILE:nash>)
endif()
